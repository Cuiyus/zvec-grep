894:     return s;

896:   // write FTS index
897:   s = insert_fts_indexer(doc);
898:   CHECK_RETURN_STATUS(s);
899:   // write vector index
900:   s = insert_vector_indexer(doc);
901:   if (!s.ok() && s != Status::AlreadyExists()) {
902:     return s;
903:   }

905:   {
906:     // Combined indexers and scalar readers copy this metadata under the
907:     // column lock. Publish the row only after its scalar/vector writes finish.
908:     std::unique_lock<std::shared_mutex> col_lock(seg_col_mtx_);
909:     auto &mem_block = segment_meta_->writing_forward_block().value();
910:     mem_block.max_doc_id_ = g_doc_id;
911:     mem_block.doc_count_ = mem_block.doc_count_ + 1;
912:   }
913:
